% Alerted: these are hazards, and NASA red is the deck's one warning ink.
\begin{alertblock}{One Silent Trap per Archive}
\scriptsize
\begin{itemize}\itemsep0.34em
  \item \textbf{ERA5} \texttt{sd} is snow \emph{water equivalent}, not depth.
        Geometric depth is \texttt{sd\,/\,rsn}
  \item \textbf{MERRA-2} \texttt{SNODP} is depth \emph{within} the snow-covered
        fraction; the grid mean is \texttt{SNODP\,$\times$\,FRSNO}
  \item \textbf{ARCO-ERA5} the time axis runs to 2050. Absent chunks read back
        as NaN, with no error
  \item \textbf{SNOTEL} every station sits above the domain's median cell.
        Timing only, never magnitude
\end{itemize}
\end{alertblock}
